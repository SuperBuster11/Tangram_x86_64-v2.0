if(WIN32)
  add_executable(tangram WIN32 src/main_win32.cpp assets/tangram.rc)   # the .rc embeds the icon
  target_link_libraries(tangram tggame winmm)
  # Tangram_x64.exe for a 64-bit build, Tangram_x86.exe for a 32-bit one
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set_target_properties(tangram PROPERTIES OUTPUT_NAME "Tangram_x64")
  else()
    set_target_properties(tangram PROPERTIES OUTPUT_NAME "Tangram_x86")
  endif()
  # builds made before the rename left tangram.exe (and its .pdb/.ilk) in the same folder; CMake never deletes old files, so do it here
  add_custom_command(TARGET tangram POST_BUILD COMMAND ${CMAKE_COMMAND} -E ${TG_COPY_DIR} "${CMAKE_SOURCE_DIR}/data" "$<TARGET_FILE_DIR:tangram>/data")   # game data (+ music*.wav)
  add_custom_command(TARGET tangram POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E rm -f "$<TARGET_FILE_DIR:tangram>/tangram.exe" "$<TARGET_FILE_DIR:tangram>/tangram.pdb" "$<TARGET_FILE_DIR:tangram>/tangram.ilk"
    VERBATIM)
else()
  find_package(X11)
  if(X11_FOUND)
    add_executable(tangram_x11 src/main_x11.cpp src/audio_linux.cpp)   # sound: libpulse-simple / libasound loaded at run time (dlopen)
    target_include_directories(tangram_x11 PRIVATE ${X11_INCLUDE_DIR})
    target_link_libraries(tangram_x11 tggame ${X11_LIBRARIES} ${CMAKE_DL_LIBS} Threads::Threads)
    add_custom_command(TARGET tangram_x11 POST_BUILD COMMAND ${CMAKE_COMMAND} -E ${TG_COPY_DIR} "${CMAKE_SOURCE_DIR}/data" "$<TARGET_FILE_DIR:tangram_x11>/data")   # game data
  endif()
endif()
