cmake_minimum_required(VERSION 3.10)
project(tangram_cpp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Tests and tools are built only for Debug (and for builds without a type); a Release build makes just the game.
# Override with -DTANGRAM_BUILD_TESTS=ON / OFF.
if(CMAKE_BUILD_TYPE MATCHES "^(Release|RelWithDebInfo|MinSizeRel)$")
  set(TG_TESTS_DEFAULT OFF)
else()
  set(TG_TESTS_DEFAULT ON)
endif()
option(TANGRAM_BUILD_TESTS "build the tests and the analysis tools" ${TG_TESTS_DEFAULT})

# The game goes to bin/ (32-bit) or bin64/ (64-bit) in the project folder, with its data/ folder next to it; everything else
# the build makes (objects, libraries, CMake files) stays in the build folder (out/build/... in Visual Studio).
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(TG_BIN_DIR "${CMAKE_SOURCE_DIR}/bin64")
else()
  set(TG_BIN_DIR "${CMAKE_SOURCE_DIR}/bin")
endif()
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.26)
  set(TG_COPY_DIR copy_directory_if_different)   # the music*.wav files are large: copy only what changed
else()
  set(TG_COPY_DIR copy_directory)
endif()
function(tg_game_output target)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${TG_BIN_DIR}" PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  # bin/ may have been deleted since CMake was configured: make it again before linking
  add_custom_command(TARGET ${target} PRE_LINK COMMAND ${CMAKE_COMMAND} -E make_directory "${TG_BIN_DIR}")
  foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)   # multi-config generators would add a Release/ subfolder otherwise
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${cfg} "${TG_BIN_DIR}" PDB_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}")
  endforeach()
  # v2.0: the game data is not compiled in - data/ is copied next to the program after every build
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E ${TG_COPY_DIR} "${CMAKE_SOURCE_DIR}/data" "$<TARGET_FILE_DIR:${target}>/data")
endfunction()

# ---- the game ----
find_package(Threads REQUIRED)   # the filters split rows over threads
add_library(tgcore src/adf.cpp src/image.cpp src/scale_filter.cpp)
target_link_libraries(tgcore Threads::Threads)
add_library(tglevels src/levels.cpp src/raster.cpp src/pieces.cpp src/board.cpp src/scoring.cpp src/passwords.cpp src/highscores.cpp src/assets.cpp src/dat.cpp src/font.cpp src/sprite.cpp)
target_link_libraries(tglevels tgcore)
add_library(tgmusic src/m68k.cpp src/music.cpp)
target_link_libraries(tgmusic tgcore)
add_library(tgassets src/data_files.cpp)
target_link_libraries(tgassets tgcore)
add_library(tggame src/game.cpp)
target_link_libraries(tggame tglevels tgmusic tgassets)

if(WIN32)
  add_executable(tangram WIN32 src/main_win32.cpp assets/tangram.rc)   # the .rc embeds the icon
  target_link_libraries(tangram tggame winmm)
  # Tangram_x64.exe for a 64-bit build, Tangram_x86.exe for a 32-bit one
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set_target_properties(tangram PROPERTIES OUTPUT_NAME "Tangram_x64")
  else()
    set_target_properties(tangram PROPERTIES OUTPUT_NAME "Tangram_x86")
  endif()
  tg_game_output(tangram)
else()
  find_package(X11)
  if(X11_FOUND)
    add_executable(tangram_x11 src/main_x11.cpp src/audio_linux.cpp)   # sound: libpulse-simple / libasound loaded at run time (dlopen)
    target_include_directories(tangram_x11 PRIVATE ${X11_INCLUDE_DIR})
    target_link_libraries(tangram_x11 tggame ${X11_LIBRARIES} ${CMAKE_DL_LIBS} Threads::Threads)
    tg_game_output(tangram_x11)
  endif()
endif()
